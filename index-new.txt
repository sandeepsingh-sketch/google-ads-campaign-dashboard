const fmtInr = n => "₹" + Number(n).toLocaleString("en-IN", {maximumFractionDigits:0});
const fmtNum = n => Number(n).toLocaleString("en-IN");
const fmtPct = n => Number(n).toFixed(2) + "%";

function renderKpis(){
  const totals = campaignData.reduce((a,r)=>{
    a.clicks+=r.clicks; a.impr+=r.impr; a.cost+=r.cost; return a;
  }, {clicks:0, impr:0, cost:0});
  const ctr = totals.impr ? (totals.clicks/totals.impr*100) : 0;
  const cpc = totals.clicks ? (totals.cost/totals.clicks) : 0;
  const cards = [
    {label:"Clicks", value: fmtNum(totals.clicks)},
    {label:"Impressions", value: fmtNum(totals.impr)},
    {label:"Cost", value: fmtInr(totals.cost)},
    {label:"CTR", value: fmtPct(ctr)},
    {label:"Avg. CPC", value: fmtInr(cpc.toFixed(0))},
  ];
  document.getElementById("kpiGrid").innerHTML = cards.map(c =>
    `<div class="kpi"><div class="label">${c.label}</div><div class="value">${c.value}</div></div>`
  ).join("");
}

function renderCampaignSection(){
  const rows = campaignData;
  const tbody = document.querySelector("#campTable tbody");
  tbody.innerHTML = rows.map(r => `<tr>
    <td class="campaign-name">${r.name}${r.status==="paused" ? ' <span class="chip paused">Paused</span>' : ''}</td>
    <td class="num">${fmtNum(r.clicks)}</td>
    <td class="num">${fmtNum(r.impr)}</td>
    <td class="num">${fmtPct(r.ctr)}</td>
    <td class="num">${fmtInr(r.cost)}</td>
  </tr>`).join("");

  const ctx = document.getElementById("campChart");
  const chartData = {
    labels: rows.map(r=>r.name),
    datasets: [
      { label: "Clicks", data: rows.map(r=>r.clicks), backgroundColor: "#1a56db", yAxisID: "y" },
      { label: "Cost (₹)", data: rows.map(r=>r.cost), backgroundColor: "#fbbc04", yAxisID: "y1" }
    ]
  };
  if (campChart) { campChart.data = chartData; campChart.update(); }
  else {
    campChart = new Chart(ctx, {
      type: "bar",
      data: chartData,
      options: {
        responsive: true, maintainAspectRatio: false,
        scales: {
          y: { position:"left", beginAtZero:true, title:{display:true,text:"Clicks"} },
          y1: { position:"right", beginAtZero:true, grid:{drawOnChartArea:false}, title:{display:true,text:"Cost (₹)"} }
        },
        plugins:{ legend:{ position:"bottom" } }
      }
    });
  }
}

function statusChip(status){
  if (status === "Paused") return `<span class="chip paused">Paused</span>`;
  if (status && status.indexOf("Not eligible") === 0) return `<span class="chip low">${status.replace("Not eligible - ","")}</span>`;
  return `<span class="chip eligible">Eligible</span>`;
}

function renderKeywordTable(campaign){
  campaign = campaign || currentKwCampaign;
  const data = keywordDataByCampaign[campaign];
  const tbody = document.querySelector("#kwTable tbody");
  tbody.innerHTML = data.rows.map(k => `<tr>
    <td>${k.kw}</td>
    <td>${k.match}</td>
    <td>${statusChip(k.status)}</td>
    <td class="num">${fmtNum(k.clicks)}</td><td class="num">${fmtNum(k.impr)}</td>
    <td class="num">${fmtPct(k.ctr)}</td><td class="num">${k.cpc?fmtInr(k.cpc.toFixed(0)):"—"}</td>
    <td class="num">${fmtInr(k.cost)}</td>
  </tr>`).join("");
  let suffix = ", including 0 impression/0 click keywords.";
  if (campaign === "Chatbot") suffix = " — campaign is paused, all keywords show 0 activity.";
  if (campaign === "Ai consulting") suffix = " — new campaign launched 17 Aug 2026, no activity yet.";
  const dateLabel = campaign === "Ai consulting" ? "verified 17 Aug 2026" : "last verified 23 Jul – 10 Aug 2026 snapshot";
  document.getElementById("kwNote").textContent =
    `Showing ${data.rows.length} keyword${data.rows.length===1?"":"s"} in ${campaign} (${dateLabel})` + suffix;
}

      options: { indexAxis:"y", responsive:true, maintainAspectRatio:false, plugins:{legend:{display:false}} }
    });
  }
}

function renderDemographics(){
  const ageCtx = document.getElementById("ageChart");
  const ageChartData = {
    labels: ageData.map(a=>a.label),
    datasets: [{ label:"Clicks", data: ageData.map(a=>a.clicks), backgroundColor:"#34a853" }]
  };
  if (ageChart) { ageChart.data = ageChartData; ageChart.update(); }
  else {
    ageChart = new Chart(ageCtx, {
      type:"bar", data: ageChartData,
      options: { responsive:true, maintainAspectRatio:false, plugins:{legend:{display:false}, title:{display:true,text:"Clicks by age"}} }
    });
  }

  const gCtx = document.getElementById("genderChart");
  const gChartData = {
    labels: genderData.map(g=>g.label),
    datasets: [{ data: genderData.map(g=>g.clicks), backgroundColor:["#ea4335","#1a56db","#9aa0a6"] }]
  };
  if (genderChart) { genderChart.data = gChartData; genderChart.update(); }
  else {
    genderChart = new Chart(gCtx, {
      type:"doughnut", data: gChartData,
      options: { responsive:true, maintainAspectRatio:false, plugins:{legend:{position:"bottom"}, title:{display:true,text:"Clicks by gender"}} }
    });
  }
}

  renderLocations();
  renderDemographics();

  renderKeywordTable(currentKwCampaign);
});
